Function chkTime(ByRef D_CycleTime As Dictionary, ByRef D As Dictionary, ByRef D_Sisaku As Dictionary, ByRef D_kumitori As Dictionary) As String

Dim i As Long
Dim tmp
Dim strOut As String
Dim x As Double
Dim x_sisaku As Double
Dim NotFound As String
Dim koteiTime_sisaku As Double
Dim Flag_nonKumitori As Boolean
koteiTime_sisaku = 0
NotFound = ""
Dim KARI As Double
Dim KARI_sisaku As Double

'量産品でサイクルタイムが無かった場合の仮の秒数
KARI = 56
'試作品でサイクルタイムが無かった場合の仮の秒数
KARI_sisaku = 75
'※片面の場合実際の生産時間は短くなる。暫定値なのでしょうがない。ちゃんと計算したい場合サイクルタイムシートに自分で入れよう！


For Each tmp In D
    Flag_nonKumitori = False
    
    If D_Sisaku.Exists(tmp) Then
        '試作なら固定時間増やす
        koteiTime_sisaku = koteiTime_sisaku + D_Sisaku(tmp)(0)
    End If
    If D_CycleTime.Exists(tmp) And VarType(D_CycleTime(tmp)) = 5 Then
        'サイクルタイムがあるなら
        If D_Sisaku.Exists(tmp) Then
            '試作の時間計算
            x_sisaku = x_sisaku + D(tmp) * D_CycleTime(tmp) + 600
        Else
            '量産品の時間計算
            x = x + D(tmp) * D_CycleTime(tmp) + 600
        End If
    Else
        'サイクルタイムDBに無いなら
        'サイクルタイムがない場合の時間計算(仮で28秒とする)
        If D_kumitori.Exists(tmp) Then
            '組取り数があったら
            If D_Sisaku.Exists(tmp) Then
                x_sisaku = x_sisaku + D(tmp) / D_kumitori(tmp) * KARI_sisaku + 600
            Else
                x = x + D(tmp) / D_kumitori(tmp) * KARI + 600
            End If
        Else
            '組取り数が無かったら
            If D_Sisaku.Exists(tmp) Then
                If D_Sisaku(tmp)(1) <> "" Then
                    x_sisaku = x_sisaku + D(tmp) / D_Sisaku(tmp)(1) * KARI_sisaku + 600
                Else
                    x_sisaku = x_sisaku + D(tmp) * KARI_sisaku + 600
                    Flag_nonKumitori = True
                End If
            Else
                x = x + D(tmp) * 28 + 600
                Flag_nonKumitori = True
            End If
        End If
        
        'サイクルタイムなしリストに品目名追加
        If NotFound = "" Then
            If Flag_nonKumitori Then
                NotFound = "★" & tmp
            Else
                NotFound = tmp
            End If
        Else
            If Flag_nonKumitori Then
                NotFound = NotFound & "|★" & tmp
            Else
                NotFound = NotFound & "|" & tmp
            End If
        End If
        
    End If
Next tmp

chkTime = x / 3600 / 0.7 & "," & NotFound & "," & koteiTime_sisaku + x_sisaku / 3600 / 0.7


End Function


Function Cal_Capa(ByRef D As Dictionary) As String

Dim HINMOKU As Dictionary
Set HINMOKU = New Dictionary
Dim tmp As Variant
Dim i As Long
Dim cnt As Double
Dim NonHit As String

Dim fso As FileSystemObject
Set fso = New FileSystemObject

Dim TMP_Path As String
TMP_Path = fso.GetSpecialFolder(TemporaryFolder).Path & "\部品リスト.accdb"


If fso.FileExists(TMP_Path) Then
    If fso.GetFile(DB_Path).DateLastModified <> fso.GetFile(TMP_Path).DateLastModified Then
        Call fso.DeleteFile(TMP_Path)
        Call fso.CopyFile(DB_Path, TMP_Path)
    End If
Else
    Call fso.CopyFile(DB_Path, TMP_Path)
End If

Dim con As ADODB.Connection
Dim rs As ADODB.Recordset

Set con = New ADODB.Connection

con.Mode = adModeRead
con.ConnectionString = "Provider=Microsoft.ACE.OLEDB.16.0;" _
                        & "Data Source=" & TMP_Path & ""
con.Open


Dim strSQL As String

'品目リスト取得
For Each tmp In D
    '存在チェック
    Set rs = con.Execute("SELECT DISTINCT 品目名 FROM egproBOM WHERE 品目名='" & tmp & "';")
    If rs.EOF Then
        If NonHit = "" Then
            NonHit = tmp
        Else
            NonHit = NonHit & "|" & tmp
        End If

    End If
    If tmp <> "" And Not rs.EOF Then
        If strSQL = "" Then
            strSQL = "'" & tmp & "'"
        Else
            strSQL = strSQL & ",'" & tmp & "'"
        End If
    End If
Next tmp

If strSQL = "" Then strSQL = "'Dummy'"

rs.Close
Set rs = Nothing


Dim DBG_3H As Double
Dim DBG_other3H As Double
Dim cnt_Tray As Long
Dim BUHINSU As Long

Set rs = con.Execute("SELECT Count([filteredBOM].[部品名]) AS 部品数, Sum(IIf( [PartList].[供給種別] = 'トレイ', 1, 0 )) AS トレイ部品数, Sum(IIf( [PartList].[三ノズルヘッド] = 1, [PartList].[フィーダーサイズ], 0 )) AS 三ノズルヘッド容量, Sum(IIf( [PartList].[三ノズルヘッド] = 0, [PartList].[フィーダーサイズ], 0 )) AS 三ノズルヘッド以外容量 FROM ( SELECT DISTINCT 部品名 FROM [egproBOM] WHERE 品目名 IN(" & strSQL & ") ) AS filteredBOM INNER JOIN PartList ON  [filteredBOM].[部品名] = [PartList].[部品名]")
If rs!部品数 = 0 Then
    DBG_3H = 0
    DBG_other3H = 0
    cnt_Tray = 0
    BUHINSU = 0
Else
    DBG_3H = rs!三ノズルヘッド容量
    DBG_other3H = rs!三ノズルヘッド以外容量
    cnt_Tray = rs!トレイ部品数
    BUHINSU = rs!部品数
End If
rs.Close
con.Close
Set rs = Nothing
Set con = Nothing

Cal_Capa = 90# - DBG_3H & "," & 300# - DBG_other3H & "," & NonHit & "," & cnt_Tray & "," & BUHINSU

Function Cal_Duplicate(ByRef FirstStr As String, SecondStr As String) As String

Dim fso As FileSystemObject
Set fso = New FileSystemObject

Dim DB_Path As String
DB_Path = "\\nksv523.ad.nkc.co.jp\第二工場実装課用\1係用\Job作成\部品リスト.accdb"

Dim TMP_Path As String
TMP_Path = fso.GetSpecialFolder(TemporaryFolder).Path & "\部品リスト.accdb"


If fso.FileExists(TMP_Path) Then
    If fso.GetFile(DB_Path).DateLastModified <> fso.GetFile(TMP_Path).DateLastModified Then
        Call fso.DeleteFile(TMP_Path)
        Call fso.CopyFile(DB_Path, TMP_Path)
    End If
Else
    Call fso.CopyFile(DB_Path, TMP_Path)
End If

Dim con As ADODB.Connection
Dim rs As ADODB.Recordset

Set con = New ADODB.Connection

con.Mode = adModeRead
con.ConnectionString = "Provider=Microsoft.ACE.OLEDB.16.0;" _
                        & "Data Source=" & TMP_Path & ""
con.Open

    Dim KyotuSlot As Double
    Dim SenyoSlot As Double
    
    Set rs = con.Execute( _
        "SELECT " & _
        "    SUM(IIf(Flag = 1, フィーダーサイズ, 0)) AS 共通部品スロット数, " & _
        "    SUM(IIf(flag Is Null, フィーダーサイズ, 0)) As 専用部品スロット数 " & _
        "From " & _
        "    ( " & _
        "        ( " & _
        "            SELECT DISTINCT " & _
        "                部品名 " & _
        "            From " & _
        "                [egproBOM] " & _
        "            WHERE " & _
        "                品目名 IN (" & FirstStr & ") " & _
        "        ) AS FirstBOM " & _
        "        LEFT JOIN ( " & _
        "            SELECT DISTINCT " & _
        "                部品名, " & _
        "                1 AS Flag " & _
        "            From " & _
        "                [egproBOM] " & _
        "            WHERE " & _
        "                品目名 IN (" & SecondStr & ") " & _
        "        ) AS SecondBOM ON firstBOM.部品名 = SecondBOM.部品名 " & _
        "    ) " & _
        "INNER JOIN PartList ON [firstBOM].[部品名] = [PartList].[部品名]; ")
    
    If IsNull(rs!共通部品スロット数) Then KyotuSlot = 0 Else KyotuSlot = rs!共通部品スロット数
    If IsNull(rs!専用部品スロット数) Then SenyoSlot = 0 Else SenyoSlot = rs!専用部品スロット数

rs.Close
con.Close
Set rs = Nothing
Set con = Nothing

Cal_Duplicate = KyotuSlot & "," & SenyoSlot

End Function

Sub Reflection_Junbi()
Dim i As Long
Sheet_Junbi.Activate
For i = 2 To Cells(Rows.Count, 1).End(xlUp).Row
    If InStr(Cells(i, 2).Formula, "XLOOKUP") <> 0 Then
        Cells(i, 2).Resize(1, 4).Copy
        Cells(i, 2).Resize(1, 4).PasteSpecial Paste:=xlPasteValues
        Application.CutCopyMode = False
    End If
    If Cells(i, 1).Text = Format(Now, "yyyy/mm/dd") Then
        Rows(i).Select
        Exit Sub
    End If
Next i
End Sub


Private Sub Worksheet_Activate()
Dim i As Long
Sheet_Junbi.Activate
For i = 2 To Cells(Rows.Count, 1).End(xlUp).Row
    If Cells(i, 1).Text = Format(Now, "yyyy/mm/dd") Then
        Rows(i).Select
        Exit Sub
    End If
Next i
End Sub

VBA MACRO CalFee.bas 
in file: xl/vbaProject.bin - OLE stream: 'VBA/CalFee'
- - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - 
Option Explicit


Function Cal_Fee(ByRef D As Dictionary) As String

Dim HINMOKU As Dictionary
Set HINMOKU = New Dictionary
Dim tmp As Variant
Dim i As Long
Dim cnt As Double
Dim NonHit As String

Dim fso As FileSystemObject
Set fso = New FileSystemObject

Dim DB_Path As String
DB_Path = "\\nksv523.ad.nkc.co.jp\第二工場実装課用\1係用\Job作成\部品リスト.accdb"

Dim TMP_Path As String
TMP_Path = fso.GetSpecialFolder(TemporaryFolder).Path & "\部品リスト.accdb"


If fso.FileExists(TMP_Path) Then
    If fso.GetFile(DB_Path).DateLastModified <> fso.GetFile(TMP_Path).DateLastModified Then
        Call fso.DeleteFile(TMP_Path)
        Call fso.CopyFile(DB_Path, TMP_Path)
    End If
Else
    Call fso.CopyFile(DB_Path, TMP_Path)
End If




Dim con As ADODB.Connection
Dim rs As ADODB.Recordset
Dim rs2 As ADODB.Recordset

Set con = New ADODB.Connection

con.Mode = adModeRead
con.ConnectionString = "Provider=Microsoft.ACE.OLEDB.16.0;" _
                        & "Data Source=" & TMP_Path & ""
con.Open


Dim strSQL As String


'有鉛DB作成
Dim D_PB As Dictionary
Set D_PB = New Dictionary
Dim cnt_PB As Long
cnt_PB = 0

With Sheet_PB
For i = 1 To .Cells(.Rows.Count, 1).End(xlUp).Row
    If Not D_PB.Exists(CStr(.Cells(i, 1).Value)) Then
        D_PB.Add CStr(.Cells(i, 1).Value), True
    End If
Next i
End With

'品目リスト取得
For Each tmp In D
    '有鉛チェック
    If D_PB.Exists(tmp) Then
        cnt_PB = cnt_PB + 1
    End If
    '存在チェック
    Set rs = con.Execute("SELECT DISTINCT 品目名 FROM egproBOM WHERE 品目名='" & tmp & "';")
    If rs.EOF Then
        If NonHit = "" Then
            NonHit = tmp
        Else
            NonHit = NonHit & "|" & tmp
        End If
        
    End If
    If tmp <> "" Or Not rs.EOF Then
        If strSQL = "" Then
            strSQL = "'" & tmp & "'"
        Else
            strSQL = strSQL & ",'" & tmp & "'"
        End If
    End If
Next tmp

rs.Close
Set rs = Nothing


'JOBの部品リストを辞書型で作成。ITEMに詳細情報をClassで作成。
Dim JOB_HINMOKU As Dictionary
Set JOB_HINMOKU = New Dictionary
Dim myBUHIN As BUHIN_Class
Dim HINMOKU_SIZE As Double
Dim BUHINSU As Long

Set rs = con.Execute("SELECT DISTINCT egproBOM.部品名, PartList.供給種別, PartList.ノズル, PartList.フィーダーサイズ, PartList.テープ幅 " & _
                     "FROM egproBOM LEFT JOIN PartList ON (egproBOM.部品名 = PartList.部品名)" & _
                     "WHERE egproBOM.品目名 IN (" & strSQL & ");")


Dim DBG_3H As Double
Dim DBG_other3H As Double

Do Until rs.EOF

    Set myBUHIN = New BUHIN_Class
    myBUHIN.部品名 = CStr(rs.Fields("部品名"))
    If IsNull(rs.Fields("供給種別")) Then
        myBUHIN.供給種別 = ""
        myBUHIN.ノズル = ""
        myBUHIN.フィーダーサイズ = 0
    Else
        myBUHIN.供給種別 = CStr(rs.Fields("供給種別"))
        myBUHIN.ノズル = CStr(rs.Fields("ノズル"))
        myBUHIN.フィーダーサイズ = CDbl(rs.Fields("テープ幅"))
    End If
    JOB_HINMOKU.Add CStr(rs.Fields("部品名")), myBUHIN
    
    HINMOKU_SIZE = HINMOKU_SIZE + CDbl(myBUHIN.フィーダーサイズ)
    BUHINSU = BUHINSU + 1
        
    Dim str
    For Each str In Split(CStr(myBUHIN.ノズル), ",")
        If CLng(str) > 1000 Then
            DBG_3H = DBG_3H + CDbl(myBUHIN.フィーダーサイズ)
            Exit For
        End If
        'If CLng(str) < 1000 Then
        '    DBG_other3H = DBG_other3H + CDbl(rs.Fields("フィーダーサイズ"))
        '    Exit For
        'End If
    Next str
    
    rs.MoveNext
Loop

Application.ScreenUpdating = False
Application.EnableEvents = False

For Each tmp In JOB_HINMOKU
    Sheet_Cal2.Cells(Rows.Count, 14).End(xlUp).Offset(1, 0).Value = tmp
    Sheet_Cal2.Cells(Rows.Count, 15).End(xlUp).Offset(1, 0).Value = JOB_HINMOKU(tmp).フィーダーサイズ
    DoEvents
Next tmp
Application.ScreenUpdating = True
Application.EnableEvents = True

rs.Close
Set rs = Nothing
Set rs2 = Nothing

Cal_Fee = 90# - DBG_3H & "," & 300# - (HINMOKU_SIZE - DBG_3H) & "," & NonHit & "," & cnt_PB & "," & BUHINSU

End Function


Sub Cal_Fee_btn()

Dim D As Dictionary
Set D = New Dictionary
Dim tmp As Variant

Columns(1).Interior.Pattern = xlNone

For Each tmp In Selection
    If tmp.Column = 1 And tmp.Value <> "" Then
        D.Add CStr(tmp.Value), tmp.Value
        tmp.Interior.Color = 65535
    End If
Next tmp

Dim str As String

If D.Count = 0 Then Exit Sub

str = Cal_Fee(D)

Cells(4, 9).Value = Split(str, ",")(0)
Cells(6, 9).Value = Split(str, ",")(1)
Cells(8, 9).Value = Split(str, ",")(2)
End Sub
