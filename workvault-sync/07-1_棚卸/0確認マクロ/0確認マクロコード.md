Public Sub CSV取り込み()
    Dim f As Variant, wb As Workbook, src As Worksheet
    Dim lastRow As Long, lastCol As Long, r As Long, c As Long
    Dim serial As String, loc As String, stockText As String
    Dim vals() As String, out() As Variant, n As Long, stock As Long
    Dim fieldInfo() As Variant, origin As Long, commitStarted As Boolean
    Dim errorNumber As Long
    Dim eDesc As String
    Dim oldLast As Long
    Dim owned As Boolean, cancelOld As XlEnableCancelKey, eventsOld As Boolean
    On Error GoTo Failed
    If Not Module1.TryBeginRun() Then Exit Sub
    owned = True
    cancelOld = Application.EnableCancelKey
    Application.EnableCancelKey = xlErrorHandler
    If Application.WorksheetFunction.CountA(Me.Columns(2)) > 0 Then
        MsgBox "B列に既存結果があるため、取り込みを中止します。", vbExclamation
        GoTo CleanExit
    End If
    If Me.Range("A1").Value2 = "シリアル番号" Then
        MsgBox "A1が原本表の見出しです。上書きせず中止します。", vbExclamation
        GoTo CleanExit
    End If
    oldLast = Me.Cells(Me.rows.Count, 1).End(xlUp).Row
    If oldLast > 1 Or Len(CStr(Me.Cells(1, 1).Value2)) > 0 Then
        If MsgBox("既存のA列を置き換えます。実行しますか。", vbYesNo + vbQuestion) <> vbYes Then GoTo CleanExit
    End If
    f = Application.GetOpenFilename("CSVファイル (*.csv),*.csv", , "取り込むCSVを選択")
    If VarType(f) = vbBoolean Then GoTo CleanExit
    If LCase$(Right$(CStr(f), 4)) <> ".csv" Then Err.Raise vbObjectError + 99, , "CSV拡張子のファイルだけを選択してください。"
    If IsWorkbookNameOpen(CStr(f)) Then Err.Raise vbObjectError + 98, , "同名のWorkbookが既に開かれています。"
    origin = CSVOrigin(CStr(f))
    ReDim fieldInfo(1 To Me.Columns.Count)
    For c = 1 To Me.Columns.Count
        fieldInfo(c) = Array(c, xlTextFormat)
    Next c
    Workbooks.OpenText fileName:=CStr(f), origin:=origin, startRow:=1, DataType:=xlDelimited, TextQualifier:=xlTextQualifierDoubleQuote, ConsecutiveDelimiter:=False, Tab:=False, Semicolon:=False, Comma:=True, Space:=False, Other:=False, fieldInfo:=fieldInfo, TrailingMinusNumbers:=False, Local:=False
    Set wb = ActiveWorkbook
    If wb Is ThisWorkbook Then
        Set wb = Nothing
        Err.Raise vbObjectError + 100, , "CSVを別ブックとして開けませんでした。"
    End If
    Set src = wb.Worksheets(1)
    lastRow = LastUsedRow(src): lastCol = LastUsedColumn(src)
    If lastRow < 1 Or lastCol < 11 Then Err.Raise vbObjectError + 101, , "CSVは11列必要です。"
    If lastCol > 11 Then
        For r = 1 To lastRow
            For c = 12 To lastCol
                If Len(CStr(src.Cells(r, c).Value2)) > 0 Then Err.Raise vbObjectError + 102, , "12列目以降に値があります。"
            Next c
        Next r
    End If
    ValidateHeaders src
    ReDim vals(1 To lastRow)
    For r = 2 To lastRow
        serial = Trim$(CStr(src.Cells(r, 1).Value2))
        loc = CStr(src.Cells(r, 8).Value2)
        stockText = CStr(src.Cells(r, 6).Value2)
        If Left$(loc, 3) = "0確認" Then
            If IsNumeric(stockText) Then
                If Not TryCLng(stockText, stock) Then Err.Raise vbObjectError + 103, , "在庫数が数値範囲外です。"
                If stock <= 9 Then
                    If Len(serial) = 0 Then Err.Raise vbObjectError + 104, , "対象行のシリアル番号が空欄です。"
                    If HasControlChars(serial) Then Err.Raise vbObjectError + 105, , "制御文字を含むシリアル番号があります。"
                    n = n + 1: vals(n) = serial
                End If
            End If
        End If
    Next r
    CloseSource wb
    If n = 0 Then
        MsgBox "対象が0件のため、既存一覧を保持しました。", vbInformation
        GoTo CleanExit
    End If
    ReDim out(1 To n, 1 To 1)
    For r = 1 To n: out(r, 1) = vals(r): Next r
    commitStarted = True
    eventsOld = Application.EnableEvents
    Application.EnableEvents = False
    Me.Columns(1).ClearContents
    Me.Range("A:A").NumberFormat = "@"
    Me.Cells(1, 1).Resize(n, 1).Value2 = out
    MsgBox n & "件を取り込みました。", vbInformation
CleanExit:
    On Error Resume Next
    If Not wb Is Nothing Then wb.Close SaveChanges:=False
    If commitStarted Then Application.EnableEvents = eventsOld
    If owned Then
        Application.EnableCancelKey = cancelOld
        Module1.EndRun
        owned = False
    End If
    Exit Sub
Failed:
    errorNumber = Err.Number: eDesc = Err.Description
    On Error Resume Next
    If Not wb Is Nothing Then wb.Close SaveChanges:=False
    If commitStarted Then Application.EnableEvents = eventsOld
    If owned Then
        Application.EnableCancelKey = cancelOld
        Module1.EndRun
        owned = False
    End If
    On Error GoTo 0
    If commitStarted Then
        MsgBox "CSV取り込みを中止しました。書き込み開始後にエラーが発生したため、既存データの復元が必要です。" & vbCrLf & "エラー番号: " & CStr(errorNumber) & vbCrLf & eDesc, vbCritical
    Else
        MsgBox "CSV取り込みを中止しました。既存一覧は変更していません。" & vbCrLf & "エラー番号: " & CStr(errorNumber) & vbCrLf & eDesc, vbExclamation
    End If
End Sub

Private Sub ValidateHeaders(ByVal ws As Worksheet)
    Dim h As Variant, i As Long
    h = Array("シリアル番号", "品番", "発注番号", "入荷日時", "受入数", "在庫数", "ステータス名称", "ロケーション名称", "棚", "棚分類コード", "最終更新日時")
    For i = 0 To 10
        If CStr(ws.Cells(1, i + 1).Value2) <> h(i) Then Err.Raise vbObjectError + 106, , "11列ヘッダが一致しません。"
    Next i
End Sub

Private Function IsWorkbookNameOpen(ByVal fileName As String) As Boolean
    Dim x As Workbook
    For Each x In Application.Workbooks
        If StrComp(x.Name, Dir$(fileName), vbTextCompare) = 0 Then
            IsWorkbookNameOpen = True
            Exit Function
        End If
    Next x
End Function

Private Function CSVOrigin(ByVal fileName As String) As Long
    Dim f As Integer, b(0 To 2) As Byte, n As Long, en As Long, ed As String
    On Error GoTo Bad
    f = FreeFile: Open fileName For Binary Access Read As #f
    n = LOF(f)
    If n >= 3 Then Get #f, 1, b
    Close #f
    If b(0) = &HEF And b(1) = &HBB And b(2) = &HBF Then CSVOrigin = 65001 Else CSVOrigin = 932
    Exit Function
Bad:
    en = Err.Number: ed = Err.Description
    On Error Resume Next
    If f <> 0 Then Close #f
    On Error GoTo 0
    Err.Raise en, "CSVOrigin", ed
End Function

Private Sub CloseSource(ByRef wb As Workbook)
    Dim en As Long, ed As String
    On Error GoTo Bad
    wb.Close SaveChanges:=False
    Set wb = Nothing
    Exit Sub
Bad:
    en = Err.Number: ed = Err.Description
    On Error Resume Next
    If Not wb Is Nothing Then wb.Close SaveChanges:=False
    Set wb = Nothing
    On Error GoTo 0
    Err.Raise en, "CloseSource", ed
End Sub

Private Function TryCLng(ByVal s As String, ByRef result As Long) As Boolean
    On Error GoTo Bad
    If Len(s) = 0 Then Exit Function
    result = CLng(s): TryCLng = True
    Exit Function
Bad:
    TryCLng = False
End Function

Private Function HasControlChars(ByVal s As String) As Boolean
    Dim i As Long, k As Long
    For i = 1 To Len(s)
        k = AscW(Mid$(s, i, 1))
        If k >= 0 And k < 32 Then HasControlChars = True: Exit Function
    Next i
End Function

Private Function LastUsedRow(ByVal ws As Worksheet) As Long
    Dim q As Range: Set q = ws.Cells.Find(What:="*", After:=ws.Cells(1, 1), LookIn:=xlFormulas, LookAt:=xlPart, SearchOrder:=xlByRows, SearchDirection:=xlPrevious, MatchCase:=False, MatchByte:=False, SearchFormat:=False)
    If q Is Nothing Then LastUsedRow = 0 Else LastUsedRow = q.Row
End Function

Private Function LastUsedColumn(ByVal ws As Worksheet) As Long
    Dim q As Range: Set q = ws.Cells.Find(What:="*", After:=ws.Cells(1, 1), LookIn:=xlFormulas, LookAt:=xlPart, SearchOrder:=xlByColumns, SearchDirection:=xlPrevious, MatchCase:=False, MatchByte:=False, SearchFormat:=False)
    If q Is Nothing Then LastUsedColumn = 0 Else LastUsedColumn = q.Column
End Function

Public Function GetWindowTitle(ByRef WindowX As Long, ByRef WindowY As Long) As String
#If VBA7 Then
    Dim hWnd As LongPtr
#Else
    Dim hWnd As Long
#End If
    Dim r As RECT, n As Long, s As String
    hWnd = GetForegroundWindow()
    GetWindowRect hWnd, r
    WindowX = r.Left: WindowY = r.Top
    n = GetWindowTextLength(hWnd)
    If n <= 0 Then GetWindowTitle = "Null": Exit Function
    s = String$(n + 1, vbNullChar): n = GetWindowText(hWnd, s, Len(s))
    GetWindowTitle = Left$(s, n)
End Function

Public Sub ゼロ確認()
    Dim ws As Worksheet, lastRow As Long, lastB As Long, i As Long, x As Long, y As Long
    Dim s As String, b As String, t As String
    Dim cancelOld As XlEnableCancelKey, owned As Boolean, errDesc As String
    Dim hadUnconfirmed As Boolean, hadRecorded As Boolean
    On Error GoTo ErrorCleanup
    If Not TryBeginRun() Then Exit Sub
    owned = True
    cancelOld = Application.EnableCancelKey
    Application.EnableCancelKey = xlErrorHandler
    Set ws = Sheet1
    lastRow = LastResultRow(ws)
    lastB = LastStatusRow(ws)
    If ws.Range("A1").Value2 = "シリアル番号" Then MsgBox "A1が原本表のため中止します。", vbExclamation: GoTo Done
    If lastRow < 1 Or Len(CStr(ws.Cells(1, 1).Value2)) = 0 Then MsgBox "A列のデータがありません。", vbExclamation: GoTo Done
    If lastB > lastRow Then MsgBox "B列だけのデータを検出したため中止します。", vbExclamation: GoTo Done
    For i = 1 To lastRow
        s = Trim$(CStr(ws.Cells(i, 1).Value2)): b = CStr(ws.Cells(i, 2).Value2)
        If Len(s) = 0 Then MsgBox "A列に空欄があります。", vbExclamation: GoTo Done
        If HasControlChars(s) Then MsgBox "シリアルに制御文字があります。", vbExclamation: GoTo Done
        If Len(b) > 0 And b <> "○" And b <> "無し" And b <> "要確認" Then MsgBox "B列に許可されない値があります。", vbExclamation: GoTo Done
        If Len(b) > 0 Then hadRecorded = True
        If b = "要確認" Then hadUnconfirmed = True
    Next i
    If hadUnconfirmed Then
        MsgBox "B列に要確認が残っています。EG-Proの結果を照合するまで再実行できません。", vbExclamation
        GoTo Done
    End If
    If Not WaitMain(x, y) Then Err.Raise vbObjectError + 201, , "メイン画面を確認できません。"
    For i = 1 To lastRow
        If Len(CStr(ws.Cells(i, 2).Value2)) = 0 Then
            ws.Cells(i, 2).Value2 = "要確認"
            hadUnconfirmed = True
            If GetWindowTitle(x, y) <> MAIN_TITLE Then Err.Raise vbObjectError + 202, , "行開始時のメイン画面を確認できません。"
            SetCursorPos x + 180, y + 69
            mouse_event MOUSE_LEFTDOWN: mouse_event MOUSE_LEFTUP
            If GetWindowTitle(x, y) <> MAIN_TITLE Then Err.Raise vbObjectError + 203, , "クリック後のメイン画面を確認できません。"
            If GetWindowTitle(x, y) <> MAIN_TITLE Then Err.Raise vbObjectError + 204, , "シリアル送信前のメイン画面を確認できません。"
            SendKeys EscapeSendKeys(Trim$(CStr(ws.Cells(i, 1).Value2))), False
            DoEventsSleep 500
            If GetWindowTitle(x, y) <> MAIN_TITLE Then Err.Raise vbObjectError + 205, , "入力後のメイン画面を確認できません。"
            SendKeys "{ENTER}", False
            DoEventsSleep 500
            t = GetWindowTitle(x, y)
            If t = DIALOG_TITLE Then
                If Not SendMainEnter(x, y) Then Err.Raise vbObjectError + 206, , "ダイアログ処理後のメイン画面を確認できません。"
                ws.Cells(i, 2).Value2 = "無し"
            ElseIf t = MAIN_TITLE Then
                If Not SendMainEnter3(x, y) Then Err.Raise vbObjectError + 207, , "メイン画面の確定操作に失敗しました。"
                If GetWindowTitle(x, y) <> MAIN_TITLE Then Err.Raise vbObjectError + 208, , "最終メイン画面を確認できません。"
                ws.Cells(i, 2).Value2 = "○"
            Else
                Err.Raise vbObjectError + 209, , "未知の画面タイトルです。"
            End If
        End If
    Next i
    MsgBox "操作が終了しました。B列を確認して保存してください。○は画面操作の完了記録です。", vbInformation
Done:
    On Error Resume Next
    If owned Then
        Application.EnableCancelKey = cancelOld
        EndRun
        owned = False
    End If
    Exit Sub
ErrorCleanup:
    errDesc = Err.Description
    On Error Resume Next
    If owned Then
        Application.EnableCancelKey = cancelOld
        EndRun
        owned = False
    End If
    On Error GoTo 0
    If hadUnconfirmed Then
        MsgBox "処理を中断しました。要確認が残っている可能性があります。" & vbCrLf & errDesc, vbExclamation
    ElseIf hadRecorded Then
        MsgBox "処理を中断しました。列記録が残っている可能性があります。" & vbCrLf & errDesc, vbExclamation
    Else
        MsgBox "処理を中断しました。" & vbCrLf & errDesc, vbExclamation
    End If
End Sub

Private Function WaitMain(ByRef x As Long, ByRef y As Long) As Boolean
    Dim started As Double, t As String
    started = Timer
    Do
        t = GetWindowTitle(x, y)
        If t = MAIN_TITLE Then WaitMain = True: Exit Function
        DoEventsSleep 50
    Loop While Elapsed(started) < WAIT_TIMEOUT
End Function

Private Function SendMainEnter(ByRef x As Long, ByRef y As Long) As Boolean
    If GetWindowTitle(x, y) <> DIALOG_TITLE Then Exit Function
    SendKeys "{ENTER}", False: DoEventsSleep 500
    SendMainEnter = (GetWindowTitle(x, y) = MAIN_TITLE)
End Function

Private Function SendMainEnter3(ByRef x As Long, ByRef y As Long) As Boolean
    Dim i As Long, t As String
    For i = 1 To 3
        t = GetWindowTitle(x, y)
        If t <> MAIN_TITLE And t <> DIALOG_TITLE Then Exit Function
        SendKeys "{ENTER}", False: DoEventsSleep 500
    Next i
    SendMainEnter3 = (GetWindowTitle(x, y) = MAIN_TITLE)
End Function

Private Sub DoEventsSleep(ByVal ms As Long)
    Dim untilAt As Double: untilAt = Timer + ms / 1000#
    Do: DoEvents: Sleep 50: Loop While Elapsed(untilAt - ms / 1000#) < ms / 1000#
End Sub

Private Function Elapsed(ByVal startAt As Double) As Double
    Elapsed = Timer - startAt: If Elapsed < 0 Then Elapsed = Elapsed + 86400#
End Function

Private Function LastResultRow(ByVal ws As Worksheet) As Long
    LastResultRow = ws.Cells(ws.rows.Count, 1).End(xlUp).Row
End Function

Private Function LastStatusRow(ByVal ws As Worksheet) As Long
    LastStatusRow = ws.Cells(ws.rows.Count, 2).End(xlUp).Row
End Function

Private Function HasControlChars(ByVal s As String) As Boolean
    Dim i As Long, k As Long
    For i = 1 To Len(s)
        k = AscW(Mid$(s, i, 1))
        If k >= 0 And k < 32 Then HasControlChars = True: Exit Function
    Next i
End Function

Private Function EscapeSendKeys(ByVal s As String) As String
    Dim i As Long, ch As String, o As String
    For i = 1 To Len(s)
        ch = Mid$(s, i, 1)
        Select Case ch
        Case "{": o = o & "{{}"
        Case "}": o = o & "{}}"
        Case "+", "^", "%", "~", "(", ")", "[", "]": o = o & "{" & ch & "}"
        Case Else: o = o & ch
        End Select
    Next i
    EscapeSendKeys = o
End Function
