このPCからxlsmのVBAは書き換えられない。Excel COM が未登録
（REGDB_E_CLASSNOTREG 0x80040154）で、Excelを外から操作できないため。
VBAバイナリを直接組み替える方法もあるが、独自圧縮の書き戻しで
ブックを壊す確率のほうが高いので採らない。VBEで手作業になる。

作業は3つ。【1】がインポート、【2】【3】が貼り付け。

(A) PtrSafe が無い Declare
    Module1 の126・127・130行目、Module2 の314行目。
    79〜118行目は #If Win64 と PtrSafe で守られているのに、この4本だけ素のまま。
    このPCの Excel は 64bit（EXCEL.EXE が C:\Program Files\ 配下）なので、
    「'PtrSafe' 属性が必要です」でここが先に落ちる。

(B) 同名の公開宣言が2つ
    Type coordinate と GetCursorPos が Module1 と Module2 の両方にあり、
    標準モジュールの既定は公開なので名前が衝突する。

VBAはプロジェクト全体をコンパイルしてから実行する。
どちらか一方でも残っていれば、CSV取り込みも消し込みも一切動かない。
「取り込んでも0件」だったのは、0件だったのではなく実行されていなかったため。

'マウス
' Private にするのは Module2 と名前が衝突するため。
' PtrSafe の判定は Win64 ではなく VBA7 で見る。PtrSafe を解するかどうかは
' ポインタ長ではなく VBA のバージョンで決まるため。
Private Type coordinate
    x As Long
    y As Long
End Type

#If VBA7 Then
    Private Declare PtrSafe Function GetCursorPos Lib "user32" (lpPoint As coordinate) As Long
    Private Declare PtrSafe Function SetCursorPos Lib "user32" (ByVal x As Long, ByVal y As Long) As Long
    Private Declare PtrSafe Sub mouse_event Lib "user32" ( _
    ByVal dwFlags As Long, _
    Optional ByVal dx As Long, _
    Optional ByVal dy As Long, _
    Optional ByVal dwDate As Long, _
    Optional ByVal dwExtraInfo As Long)

    Private Declare Function SetCursorPos Lib "user32" (ByVal x As Long, ByVal y As Long) As Long

    ByVal dwFlags As Long, _

現行からの変更は2箇所だけ。判定ロジックは正しいので触っていない。

  ・A列を先に消す      前回の取り込みのほうが件数が多いと、古い行が下に残る
  ・Trim(cols(0))       29,246行のうち12,218行のシリアルが先頭に空白を持つ

- - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - - -

Private Sub Input0_Click()

    Dim filePath As String
    filePath = Application.GetOpenFilename("CSVファイル,*.csv")
    If filePath = "False" Then Exit Sub

    ' 前回の取り込みが残っていると件数が合わなくなる。B列は既存マクロの
    ' 再開位置判定に使うので、消すのはA列だけにする。
    Columns(1).ClearContents

    ' --- ファイル全体を UTF-8 で一括読み込み（BOMは自動除去） ---
    Dim st As Object
    Set st = CreateObject("ADODB.Stream")
    st.Charset = "UTF-8"
    st.Open
    st.LoadFromFile filePath
    Dim allText As String
    allText = st.ReadText(-1)   ' -1 = adReadAll ：全文を一度に読む
    st.Close

    ' --- 改行コードを自前で正規化してから行分割 ---
    allText = Replace(allText, vbCrLf, vbLf)   ' CRLF → LF
    allText = Replace(allText, vbCr, vbLf)     ' CR   → LF
    Dim lines() As String
    lines = Split(allText, vbLf)               ' LF で確実に行分割

    Dim writeRow As Long
    writeRow = 1

    Dim i As Long
    Dim cols() As String
    Dim loc As String

    For i = LBound(lines) To UBound(lines)
        If Len(Trim(lines(i))) > 0 Then
            cols = Split(lines(i), ",")
            If UBound(cols) >= 7 Then
                loc = Trim(cols(7))                 ' 8列目 ロケーション名称
                If Left(loc, 3) = "0確認" And IsNumeric(cols(5)) Then
                    If CLng(cols(5)) <= 9 Then
                        ' EG-Proのシリアルは左に空白が入る。そのまま書くと
                        ' リール消費画面の照合で一致しない。
                        Cells(writeRow, 1).Value = Trim(cols(0))
                        writeRow = writeRow + 1
                    End If
                End If
            End If
        End If
    Next i

    MsgBox writeRow - 1 & " 件を取り込みました。"

End Sub

  1. 【1】〜【3】を直し、【4】でコンパイルを通す
  2. 取り込みを実行し、5,848件と表示されるか見る
