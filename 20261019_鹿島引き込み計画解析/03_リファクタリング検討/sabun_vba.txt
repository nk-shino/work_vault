Sub do_prepare(control As IRibbonControl)
    Dim i As Long
    Dim count As Long
    Dim rng As Range
    
    Dim wb As Workbook
    Set wb = ThisWorkbook
    
    
    Dim wsJobs As Worksheet
    On Error Resume Next
    Set wsJobs = Workbooks("段取り予定.xlsx").Worksheets("job作成")
    On Error GoTo 0
    
    If wsJobs Is Nothing Then
        MsgBox "段取り予定を開いてください。"
        Exit Sub
    End If
    
    wsJobs.Activate
    
    If MsgBox("ジョブ名セルを自動選択しますか？" & vbCrLf & "(Noなら選択中範囲から取得)", vbYesNo) = vbYes Then
        selectJobName wsJobs
    End If
    
    If Selection.count = 0 Then Exit Sub
    Dim jobs() As Variant
    ReDim jobs(1 To Selection.count)
    
    '範囲内セルで、正規表現でジョブ名を識別してセルをjobsに格納。
    Dim rng_sch As Range
    Set rng_sch = Selection
    
    count = 0
    For Each rng In rng_sch
        If IsJobName(CStr(rng.Value)) Then
           count = count + 1
           Set jobs(count) = rng
        End If
    Next
    
    '足りない分のシートをコピー生成
    If count > 6 Then
        Sx_1.Visible = xlSheetVisible
        Sx_2.Visible = xlSheetVisible
        
        For i = 7 To count
            Sx_1.Copy After:=wb.Worksheets(wb.Worksheets.count)
            wb.Worksheets(wb.Worksheets.count).Name = "差分" & i
            Sx_2.Copy After:=wb.Worksheets(wb.Worksheets.count)
            wb.Worksheets(wb.Worksheets.count).Name = "計画" & i
        Next
        
        Sx_1.Visible = xlSheetHidden
        Sx_2.Visible = xlSheetHidden
    End If
    
    '「差分確認マクロ」ブックの差分シートにジョブ名と指図をコピー
    For i = 1 To UBound(jobs)
        If Not jobs(i) Is Nothing Then
            copy_job jobs, i
        End If
    Next i
    
    ' "差分1"シートをアクティブにして終了
    S1_1.Activate
    
    Set wb = Nothing
    Set wsJobs = Nothing
    Set rng_sch = Nothing
    Set rng = Nothing
    For i = 1 To UBound(jobs)
        Set jobs(i) = Nothing
    Next i
End Sub


Sub copy_job(job As Variant, num As Long)
    '引数に渡したjobのジョブ名と、Offsetで指図もコピー
    Dim dst As Range
    Dim i As Long
    i = 1
    Set dst = ThisWorkbook.Worksheets("差分" & num).Cells(2, 1)
    dst.Value = job(num).Value
    Do While job(num).Offset(i, 3).Value Like "#########"
        dst.Offset(i, 0).Value = job(num).Offset(i, 3).Value
        i = i + 1
    Loop
    Set dst = Nothing
End Sub

'段取り予定.job作成シートを受け取り、ジョブ名セルを選択状態にする
Sub selectJobName(ByVal ws As Worksheet)
Dim f As Range
Dim lastrow As Long
Dim rngCol As Range

Set f = ws.Rows(1).Find(What:="備考1", LookIn:=xlValues, LookAt:=xlWhole)
If f Is Nothing Then
    MsgBox "1行目に[備考1]列がありません。"
    Exit Sub
End If

lastrow = ws.Cells(ws.Rows.count, f.Column).End(xlUp).Row
Set rngCol = ws.Range(ws.Cells(2, f.Column), ws.Cells(lastrow, f.Column))

Dim rng As Range
Dim hit As Range
For Each rng In rngCol
    If IsJobName(CStr(rng.Value)) Then
        If hit Is Nothing Then
            Set hit = rng
        Else
            Set hit = Union(hit, rng)
        End If
    End If
Next


If hit Is Nothing Then
    MsgBox "ジョブ名パターンにマッチするセルがありません。"
    Exit Sub
End If

ws.Activate
hit.Select
hit.Cells(1).Activate

End Sub

'文字列がジョブ名のパターンに合致するか返す
Function IsJobName(ByVal str As String) As Boolean
IsJobName = False
    
Dim re As New RegExp
re.IgnoreCase = False
re.Global = False
re.Pattern = "^\d{6}\-[0-9A-Z]\d[1-3]$"

IsJobName = re.test(str)

End Function

'シート名数字部分をkeyとする差分シートと計画シートのペアを格納した辞書を返す
Function getDic() As Dictionary
Dim D As New Dictionary
Dim indexNum As String
Dim ws As Worksheet
Dim keikaku As Worksheet


For Each ws In ThisWorkbook.Worksheets
    If ws.Name Like "差分*" Then
        indexNum = Mid$(ws.Name, 3)
        
        If IsNumeric(indexNum) Then
            Set keikaku = Nothing
            
            On Error Resume Next
            Set keikaku = ThisWorkbook.Worksheets("計画" & indexNum)
            On Error GoTo 0
            
            If keikaku Is Nothing Then
                MsgBox "計画" & indexNum & "シートがありません。"
            Else
                Dim tmp As Variant
                tmp = Array(Nothing, Nothing)
                Set tmp(0) = ws
                Set tmp(1) = keikaku
            End If
            
            D.Add indexNum, tmp
        End If
    End If
Next

Set getDic = D

End Function

'臨時シートを削除
Sub delExSheets()
    Dim i As Long
    Dim nm As String
    Dim n As Long

    On Error GoTo cleanup
    Application.DisplayAlerts = False

    For i = ThisWorkbook.Worksheets.count To 1 Step -1
        nm = ThisWorkbook.Worksheets(i).Name

        If Left$(nm, 2) = "差分" Or Left$(nm, 2) = "計画" Then
            n = Val(Mid$(nm, 3))
            If n >= 7 Then ThisWorkbook.Worksheets(i).Delete
        End If
    Next i
    
cleanup:
    Application.DisplayAlerts = True
End Sub

Sub sabunTemplate(ByVal ws As Worksheet)
    Dim fSabun As String
    Dim fB As String, fc As String, fD As String, fE As String, fF As String, fG As String, fH As String, fI As String, fJ As String

    ' sabun シート参照
    fSabun = "sabun"

    ' MATCH(INDIRECT("'"&ws.Name&"'!$A2"), sabun!$A:$A, 0)
    fB = "=INDEX(sabun!B:B,MATCH($A2,sabun!$A:$A,0))"
    fc = "=INDEX(sabun!C:C,MATCH($A2,sabun!$A:$A,0))"
    fE = "=INDEX(sabun!D:D,MATCH($A2,sabun!$A:$A,0))"
    fF = "=INDEX(sabun!E:E,MATCH($A2,sabun!$A:$A,0))"
    fG = "=INDEX(sabun!F:F,MATCH($A2,sabun!$A:$A,0))"
    fH = "=INDEX(sabun!G:G,MATCH($A2,sabun!$A:$A,0))"
    fI = "=INDEX(sabun!H:H,MATCH($A2,sabun!$A:$A,0))"
    fJ = "=INDEX(sabun!I:I,MATCH($A2,sabun!$A:$A,0))"

    ' D列（PCB名）：
    fD = "=LET(s,REGEXREPLACE($C2,""S-"",""""),REGEXREPLACE(s,""([^\d])- "",""$1""))"

    ' ---- 見出し ----
    ws.Range("A1").Value = "指図"
    ws.Range("B1").Value = "担当"
    ws.Range("C1").Value = "品目コード"
    ws.Range("D1").Value = "PCB名"
    ws.Range("E1").Value = "台数"
    ws.Range("F1").Value = "指図登録日"
    ws.Range("G1").Value = "製造開始日"
    ws.Range("H1").Value = "完成予定日"
    ws.Range("I1").Value = "タイプ"
    ws.Range("J1").Value = "差分"

    ' ---- A列 2～100 行目：空欄 ----
    ws.Range("A2:A100").ClearContents

    ' ---- 数式（2～100 行目）----
    ws.Range("B2:B100").Formula2 = fB
    ws.Range("C2:C100").Formula2 = fc
    ws.Range("D2:D100").Formula2 = fD
    ws.Range("E2:E100").Formula2 = fE
    ws.Range("F2:F100").Formula2 = fF
    ws.Range("G2:G100").Formula2 = fG
    ws.Range("H2:H100").Formula2 = fH
    ws.Range("I2:I100").Formula2 = fI
    ws.Range("J2:J100").Formula2 = fJ
End Sub


'差分*シートの条件付き書式
Sub sabunSetCF(ByVal ws As Worksheet)
    Dim rng As Range
    Dim fc As FormatCondition

    Set rng = ws.Range("$J$2:$J$100")

    ' 既存の条件付き書式を一旦消して作り直す
    ws.Cells.FormatConditions.Delete

    ' ---- 1) DONE を含む ----
    Set fc = rng.FormatConditions.Add( _
                Type:=xlTextString, _
                String:="DONE", _
                TextOperator:=xlContains)
    With fc
        .Font.Color = RGB(0, 97, 0)
        .Interior.Color = RGB(198, 239, 206)
        .Font.Bold = False
        .StopIfTrue = False
    End With

    ' ---- 2) NG を含む ----
    Set fc = rng.FormatConditions.Add( _
                Type:=xlTextString, _
                String:="NG", _
                TextOperator:=xlContains)
    With fc
        .Font.Color = RGB(156, 0, 6)
        .Font.Bold = True
        .Interior.Color = RGB(255, 199, 206)
        .StopIfTrue = False
    End With
End Sub


'手動でまとめて数式更新
Sub prepare()
Dim ws As Worksheet
Dim D As New Dictionary
Dim key As Variant
Set D = getDic()

For Each key In D.Keys
    'sabunTemplate D(key)(0)
    sabunSetCF D(key)(0)
Next
End Sub
